7. Deploy ​

这一篇把前面 6 个中间件和 3 个 .NET 服务合成一份 compose.yaml，目标是一条命令起全套。每个中间件单独怎么装、控制台怎么点，见对应篇章。

前置：装好 Docker Engine + Compose V2 插件、.NET 10 SDK（用于本地编译）、curl。本机无 Docker 时只能做配置校验，不能实跑。

7.4 compose.yaml（完整） ​

yaml
services:
  # ---------- 基础设施 ----------
  postgres-order:
    image: postgres:16.15-alpine
    container_name: postgres-order
    environment:
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:?请在 .env 设置 POSTGRES_PASSWORD}
      POSTGRES_DB: orderdb
    ports:
      - "127.0.0.1:5432:5432"
    volumes:
      - pg-order:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U postgres -d orderdb -h 127.0.0.1"]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 30s
    restart: unless-stopped
    networks: [micro-shop]

  postgres-product:
    image: postgres:16.15-alpine
    container_name: postgres-product
    environment:
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
      POSTGRES_DB: productdb
    ports:
      - "127.0.0.1:5433:5432"
    volumes:
      - pg-product:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U postgres -d productdb -h 127.0.0.1"]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 30s
    restart: unless-stopped
    networks: [micro-shop]

  rabbitmq:
    image: rabbitmq:4.4.0-management
    container_name: rabbitmq
    hostname: rabbitmq
    environment:
      RABBITMQ_DEFAULT_USER: ${RABBITMQ_USER:?请在 .env 设置 RABBITMQ_USER}
      RABBITMQ_DEFAULT_PASS: ${RABBITMQ_PASSWORD:?请在 .env 设置 RABBITMQ_PASSWORD}
    ports:
      - "127.0.0.1:5672:5672"
      - "127.0.0.1:15672:15672"
    volumes:
      - rabbitmq-data:/var/lib/rabbitmq
    healthcheck:
      test: ["CMD", "rabbitmq-diagnostics", "-q", "check_port_connectivity"]
      interval: 30s
      timeout: 10s
      retries: 10
      start_period: 30s
    restart: unless-stopped
    networks: [micro-shop]

  consul:
    image: hashicorp/consul:1.21.3
    container_name: consul
    command: "agent -server -bootstrap-expect=1 -ui -client=0.0.0.0"
    ports:
      - "127.0.0.1:8500:8500"
    volumes:
      - consul-data:/consul/data
    healthcheck:
      test: ["CMD-SHELL", "curl -fsS http://127.0.0.1:8500/v1/status/leader | grep -q ."]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 10s
    restart: unless-stopped
    networks: [micro-shop]

  # ---------- 应用 ----------
  order-api:
    build:
      context: .
      dockerfile: src/Order.Api/Dockerfile
    image: micro-shop/order-api:dev
    environment:
      SERVICE_HOST: order-api
      SERVICE_PORT: "8080"
      DB: "Host=postgres-order;Port=5432;Database=orderdb;Username=postgres;Password=${POSTGRES_PASSWORD}"
      CONSUL: "http://consul:8500"
      MQ_HOST: rabbitmq
      MQ_USER: ${RABBITMQ_USER}
      MQ_PASSWORD: ${RABBITMQ_PASSWORD}
    depends_on:
      postgres-order:  { condition: service_healthy }
      rabbitmq:        { condition: service_healthy }
      consul:          { condition: service_healthy }
    restart: unless-stopped
    networks: [micro-shop]

  product-api:
    build:
      context: .
      dockerfile: src/Product.Api/Dockerfile
    image: micro-shop/product-api:dev
    environment:
      SERVICE_HOST: product-api
      SERVICE_PORT: "8080"
      DB: "Host=postgres-product;Port=5432;Database=productdb;Username=postgres;Password=${POSTGRES_PASSWORD}"
      CONSUL: "http://consul:8500"
      MQ_HOST: rabbitmq
      MQ_USER: ${RABBITMQ_USER}
      MQ_PASSWORD: ${RABBITMQ_PASSWORD}
    ports:
      - "127.0.0.1:5201:8080"     # 直连查库存，绕开网关 5 秒缓存
    depends_on:
      postgres-product: { condition: service_healthy }
      rabbitmq:         { condition: service_healthy }
      consul:           { condition: service_healthy }
    restart: unless-stopped
    networks: [micro-shop]

  ocelot-gateway:
    build:
      context: .
      dockerfile: src/Ocelot.Gateway/Dockerfile
    image: micro-shop/ocelot-gateway:dev
    environment:
      GATEWAY_PORT: "8080"
      OCELOT_CONFIG: ocelot.docker.json
    ports:
      - "127.0.0.1:8080:8080"
    depends_on:
      consul:      { condition: service_healthy }
      order-api:   { condition: service_started }
      product-api: { condition: service_started }
    restart: unless-stopped
    networks: [micro-shop]

volumes:
  pg-order:
  pg-product:
  rabbitmq-data:
  consul-data:

networks:
  micro-shop:
    driver: bridge

要点：

  • 所有容器内地址都写服务名（postgres-order、rabbitmq、consul、order-api），不能写 localhost。所有服务挂同一个 micro-shop 网络，Consul 才能按 order-api 反向做健康检查、网关才能解析注册回来的地址。
  • SERVICE_HOST 是注册给 Consul 的地址，必须是不带 scheme 的服务名；SERVICE_PORT 是容器内部端口 8080，不是宿主映射端口。
  • MQ 账号改成了配置注入，对应 OrderService 里 MQ_USER / MQ_PASSWORD 两行；不注入就会回退到代码里的演示默认值，跟 .env 账号对不上就连不上。
  • 应用容器没有定义 healthcheck（aspnet:10.0 镜像不保证带 curl），网关用 service_started 门控，就绪靠后面的冒烟重试兜底；若需要严格门控，先给应用镜像装 curl 再加 /health 检查。
  • product-api 额外映射 5201:8080，是为了查库存时绕开网关缓存；compose.yaml 里 image: 的 dev tag 是给 6.Jenkins 的回滚用的。

FileCacheOptions 在 24.1 起废弃、官方要求 25.0 改用 CacheOptions，别再用旧名。另外 RouteKeys: ["orders"] 表示全局限流选项只作用于 Key == "orders" 的那条路由；通配路由 /orders/{everything} 没有自己的 RateLimitOptions，因此不受限流（/orders/replay 才能连打 3 次）。ClientIdHeader: "Host" 让 curl 不必带 Oc-Client 头，但所有客户端会共用一个计数器，只适合演示。

ocelot*.json 要能进发布产物，OrderService 的网关 csproj 里那行 <Content Update="ocelot*.json" CopyToOutputDirectory="Always" /> 不能删。

7.9 排错 ​

现象原因 / 改法
应用容器一直重启DB / CONSUL / MQ_HOST 还写 localhost；docker compose logs -f order-api
Consul 里实例是红的SERVICE_HOST 必须是服务名、SERVICE_PORT=8080；确认所有服务在同一 micro-shop 网络
网关 503ocelot.docker.json 的 Host 是 consul；OCELOT_CONFIG 环境变量要设；Consul 里要有 passing 实例
网关启动报找不到 ocelot.docker.json文件放在 src/Ocelot.Gateway/ 下，且 csproj 有 CopyToOutputDirectory
下单后库存不变看 product-api 日志；确认 rabbitmq healthy、MQ 账号密码一致、订阅 topic 与 Group 一字不差
下单报 cap.published 表不存在CAP 启动建表失败或连错库；看应用日志，必要时 down -v 重建卷
查库存看到旧值走 :8080/products/1 有 5 秒缓存；查 :5201/products/1
docker compose up 很快返回但服务没起--wait 只在定义了 healthcheck 的服务上生效；用 docker compose ps 与冒烟重试
端口 8080 / 8500 被占本机还跑着 dotnet run 或旧容器；先 docker compose down 再查
容器名解析失败忘了给所有服务加 networks: [micro-shop]

7.11 与生产的距离（照这套上生产前必须补） ​

缺口建议
Database.EnsureCreated()改 EF Core Migrations；CAP 的 cap.* 表由 CAP 自建、不要写进迁移（见 3.PostgreSQL）
单节点 Consul、单节点 RabbitMQ、单实例数据库各自做集群 / 高可用；Consul 生产 3~5 server
明文密码放 .env改 Docker secrets / 密钥管理，端口只绑内网
无 TLS、无认证网关与中间件之间启用 TLS；Consul 启用 ACL；RabbitMQ 删除 guest
无资源限制、无日志轮转加 deploy.resources.limits 与 logging.options.max-size/max-file
.NET 10.0 浮动 tag钉补丁版或 digest
无监控与备份Prometheus/Grafana、日志聚合、pg_dump/consul snapshot save 并演练恢复
Compose 单机编排多主机滚动升级 / 自愈交给 Swarm、K8s 或托管容器服务；CI 见 6.Jenkins
